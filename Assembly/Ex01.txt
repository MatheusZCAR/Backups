.MODEL SMALL          ;hello world sem repetição
;mov é utilizado para mover valores na memória
.data
    msg db 'Ola, mundo :)',10,'$'
.CODE
    main proc
        mov ax,@data       ;inicialização do ds
        mov ds,ax		  
        mov ah,09h	   ;função 09 da int 21 - escreve a string	
        lea dx,msg	   
        int 21h
        mov ah,4ch
        int 21h
main endp
end main
